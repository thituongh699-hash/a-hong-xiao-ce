{先锋上车 V2 - 量价共振研究版；建议日线；未经历史回测验证}
N1:=10;
N2:=30;
R10:=(CLOSE-LLV(LOW,N1))/MAX(HHV(HIGH,N1)-LLV(LOW,N1),0.01)*100;
K10:=SMA(R10,5,1);
D10:=SMA(K10,3,1);
先锋:4*K10-3*D10,COLORCYAN,LINETHICK2;
R30:=(CLOSE-LLV(LOW,N2))/MAX(HHV(HIGH,N2)-LLV(LOW,N2),0.01)*100;
K30:=SMA(R30,5,1);
中期:SMA(K30,3,1),COLORFF6EC7,LINETHICK1;
底:5,COLORCD7F32,DOTLINE;
强线:25,COLORRED,DOTLINE;
小顶:90,COLORGREEN;
大顶:105,COLORMAGENTA;
STICKLINE(先锋>25,先锋,25,2,0),COLORRED;
前均量:=REF(MA(VOL,10),1);
量确认:=VOL>REF(HHV(VOL,9),1) AND VOL>前均量*1.30;
收盘质量:=CLOSE>OPEN AND CLOSE>REF(CLOSE,1) AND (CLOSE-LOW)/MAX(HIGH-LOW,0.01)>=0.65;
趋势确认:=CLOSE>MA(CLOSE,15) AND MA(CLOSE,15)>REF(MA(CLOSE,15),3);
动能确认:=先锋>25 AND 先锋>REF(先锋,1) AND 中期>=REF(中期,1);
涨幅控制:=CLOSE/REF(CLOSE,3)<1.17 AND CLOSE/REF(CLOSE,10)<1.50;
有效样本:=BARSCOUNT(CLOSE)>60;
原始信号:=量确认 AND 收盘质量 AND 趋势确认 AND 动能确认 AND 涨幅控制 AND 有效样本;
买入提示:=FILTER(原始信号,9);
DRAWICON(买入提示,12,1);
DRAWTEXT(买入提示,16,'量价共振'),COLORYELLOW;
