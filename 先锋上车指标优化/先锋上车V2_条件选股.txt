{同花顺条件选股专用；与V2副图的买入提示相同；建议日线}
R10:=(CLOSE-LLV(LOW,10))/MAX(HHV(HIGH,10)-LLV(LOW,10),0.01)*100;
K10:=SMA(R10,5,1);
D10:=SMA(K10,3,1);
先锋值:=4*K10-3*D10;
R30:=(CLOSE-LLV(LOW,30))/MAX(HHV(HIGH,30)-LLV(LOW,30),0.01)*100;
K30:=SMA(R30,5,1);
中期值:=SMA(K30,3,1);
量确认:=VOL>REF(HHV(VOL,9),1) AND VOL>REF(MA(VOL,10),1)*1.30;
收盘质量:=CLOSE>OPEN AND CLOSE>REF(CLOSE,1) AND (CLOSE-LOW)/MAX(HIGH-LOW,0.01)>=0.65;
趋势确认:=CLOSE>MA(CLOSE,15) AND MA(CLOSE,15)>REF(MA(CLOSE,15),3);
动能确认:=先锋值>25 AND 先锋值>REF(先锋值,1) AND 中期值>=REF(中期值,1);
涨幅控制:=CLOSE/REF(CLOSE,3)<1.17 AND CLOSE/REF(CLOSE,10)<1.50;
有效样本:=BARSCOUNT(CLOSE)>60;
原始信号:=量确认 AND 收盘质量 AND 趋势确认 AND 动能确认 AND 涨幅控制 AND 有效样本;
XG:FILTER(原始信号,9);
